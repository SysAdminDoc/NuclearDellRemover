# --- Alienware Command Center (AWCC) companion devices ---
# The AWCC RGB / AlienFX lighting controller. Blocking this prevents the
# AlienFX USB HID pipeline from reinstalling. WARNING: also breaks RGB lighting.
# Vendor 187C = Alienware. Product IDs vary by generation.
# USB\VID_187C&PID_0529
# USB\VID_187C&PID_0511


# --- Dell SupportAssist / Data Vault companions ---
# Dell Data Vault driver (telemetry companion). Safe to block on machines where
# SupportAssist has been removed; this stops Windows Update from re-pushing the
# telemetry driver and triggering SupportAssist reinstall.
# ROOT\DDVDD
# ROOT\DDVSC


# --- PC-Doctor hardware diagnostic module ---
# PC-Doctor ships with SupportAssist as a diagnostic helper driver.
# ROOT\PC-DOCTOR


# --- Dell Peripheral Manager / Dell Pair companion devices ---
# These ride along with Dell's wireless peripheral stack. Only block if you
# are NOT using Dell wireless keyboards, mice, or headsets.
# ROOT\DellPeripheralManager
# ROOT\DellPair
